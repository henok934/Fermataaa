function filterBusNames(selectedLevel) {
const nameWrapper = document.getElementById("bus-type-wrapper");
const nameSelect = document.getElementById("bus-name-select");
const plateWrapper = document.getElementById("plate-side-wrapper");
// Reset everything
nameSelect.value = "";
plateWrapper.style.display = "none";
if (selectedLevel) {
nameWrapper.style.display = "block";
let seenNames = new Set();
Array.from(nameSelect.options).forEach(option => {
if (option.value === "") return;
const matchesLevel = option.getAttribute("data-level").trim() === selectedLevel.trim();
if (matchesLevel && !seenNames.has(option.value)) {
option.style.display = "block";
option.disabled = false;
seenNames.add(option.value);
} else {
option.style.display = "none";
option.disabled = true;
}
});
} else {
nameWrapper.style.display = "none";
}
}
function filterPlatesAndSides(selectedName) {

} else {
plateWrapper.style.display = "none";

}
